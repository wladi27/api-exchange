DEVELOPER EXPERIENCE (DX)

Integración en Menos de 2 Minutos

Usa nuestro paquete oficial para NPM con TypeScript o consume los endpoints REST desde cualquier lenguaje. 

// 1. Instalar SDK: npm install @dolar-api/sdk
import { DolarApiClient } from '@dolar-api/sdk';

const client = new DolarApiClient({
  apiKey: 'bcv_live_tu_clave_secreta'
});

// Obtener tasas oficiales
const rates = await client.getRates();
console.log(`USD: ${rates.prices.usd_bs} Bs. | EUR: ${rates.prices.eur_bs} Bs.`);

// Conversión directa de divisas
const conversion = await client.convert({ amount: 100, from: 'USD', to: 'VES' });
console.log(`100 USD = ${conversion.result} Bs.`);
